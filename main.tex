\section{Leis de composição interna e externa}

\hypertarget{def111}{}
\begin{definition}
Seja $E$ um conjunto. Chamamos de \textbf{lei de composição interna sobre $E$} a função:

\[
\star\;:
\begin{array}{r c l}
  E\times E &\to &  E \\
  (x,y) &\mapsto & \star(x,y)\overset{def}{=} x\star y
\end{array}
\]

\end{definition}

\hypertarget{def112}{}
\begin{definition}
Sejam $E$ e $\Omega$ dois conjuntos. Chamamos de \textbf{lei de composição externa de $\Omega$ sobre $E$} a função:

\[
\odot\;:
\begin{array}{r c l}
  \Omega\times E &\to &  E \\
  (\lambda,x) &\mapsto & \odot(\lambda,x)\overset{def}{=} \lambda\odot x
\end{array}
\]

\end{definition}

\textit{Observação: caso não haja ambiguidade e os conjuntos estejam implícitos no contexto, podemos apenas escrever \textbf{lei de composição interna} ou \textbf{lei interna} e \textbf{lei de composição externa} ou \textbf{lei externa}.}

\begin{example}
Considere os conjuntos $\mathbb{R}$ e $\mathbb{R}^2$.

A adição usual de elementos de $\mathbb{R}^2$ é uma lei de composição interna sobre $\mathbb{R}^2$:
\[
+\;:
\begin{array}{r c l}
  \mathbb{R}^2\times \mathbb{R}^2 &\to &  \mathbb{R}^2 \\
  ( (x_1,y_1), (x_2,y_2) ) &\mapsto & (x_1+x_2, y_1+y_2)
\end{array}
\]

A multiplicação por um escalar real é uma lei de composição externa de $\mathbb{R}$ sobre $\mathbb{R}^2$:
\[
\cdot\;:
\begin{array}{r c l}
  \mathbb{R}\times \mathbb{R}^2 &\to &  \mathbb{R}^2 \\
  (\lambda, (x,y)) &\mapsto & (\lambda x, \lambda y)
\end{array}
\]
\end{example}

\begin{definition}
Uma lei de composição interna ''$\star$'' sobre um conjunto $E$ é dita:

\begin{itemize}
  \item[(i)] \textbf{associativa}, se $(x\star y)\star z = x\star(y\star z) = x\star\star z,\; \forall x,y,z\in E$
  \item[(ii)] \textbf{comutativa}, se $x\star y=y\star x,\; \forall x,y\in E$
\end{itemize}


\end{definition}

\chapter{Generalidades sobre Espaços Vetoriais}

Em todo este capítulo, $\mathbb{K}$ designa um corpo de característica diferente de $2$.

\section{Espaços Vetoriais}

\begin{definition}

Seja $E$ um conjunto não vazio e $\oplus$ e $\odot$ as operações de adição e multiplicação para $\mathbb{K}$. Dizemos que $E$ munido das leis interna e externa, respectivamente:

\[
+\;:
\begin{array}{r c l}
  E\times E &\to &  E \\
  (x,y) &\mapsto & x+y
\end{array}
\qquad \text{e,} \qquad
\cdot\;:
\begin{array}{r c l}
  \mathbb{K}\times E &\to & E \\
  (\lambda, x) &\mapsto & \lambda x
\end{array}
\]

\textbf{é um espaço vetorial sobre $\mathbb{K}$} se, e somente se, $(E, +)$ é um grupo abeliano e:

\begin{itemize}
  \item[(i)] $\lambda(\mu x) = (\lambda \odot \mu)x, \quad \forall x\in E,\; \forall \lambda,\mu\in \mathbb{K}$
  \item[(ii)] $(\lambda \oplus \mu)x = \lambda x + \mu x, \quad \forall x\in E,\; \forall \lambda,\mu\in \mathbb{K}$
  \item[(iii)] $\lambda(x+y) = \lambda x + \lambda y, \quad \forall x,y\in E,\; \forall \lambda\in \mathbb{K}$
  \item[(iv)] $1\cdot x = x, \quad \forall x\in E$
\end{itemize}

onde $1$ representa o elemento neutro para a multiplicação em $\mathbb{K}$.
\end{definition}

\textit{Observações:}

\begin{itemize}
  \item \textit{Chamamos comumente os elementos de $\mathbb{K}$ de \textbf{escalares} e os elementos de $E$ de \textbf{vetores}};
  \item \textit{Denominamos um espaço vetorial $E$ sobre $\mathbb{K}$ de $\mathbb{K}$-espaço vetorial};
  \item \textit{Caso não haja ambiguidade, denotamos comumente os mesmos símbolos usuais de soma e produto para as operações do corpo $\mathbb{K}$ e as leis interna e externa do espaço vetorial $E$};
  \item \textit{Note que a definição de espaço vetorial é equivalente a afirmar que o corpo $\mathbb{K}$ \textbf{age} sobre o grupo abeliano $E$}.
\end{itemize}

\begin{proposition}

Seja $E$ um $\mathbb{K}$-espaço vetorial. Então, $\forall (\lambda,x)\in \mathbb{K}\times E$, temos:

\begin{itemize}
  \item[(i)] $\lambda 0_E$ e $0x = 0_E$
  \item[(ii)] $\lambda x = 0_E \implies \lambda = 0$ ou $x = 0_E$
  \item[(iii)] $(-\lambda)x = \lambda(-x) = -(\lambda x)$
\end{itemize}

\end{proposition}

\textbf{Demonstração:}

Mostremos primeiramente o item $(i)$. Note que:

\[
\lambda 0_E = \lambda (0_E + 0_E) = \lambda 0_E + \lambda 0_E \implies \lambda 0_E = 0_E
\]

Analogamente, temos também que:

\[
0x = (0+0)x = 0x + 0x \implies 0x = 0_E 
\]

Note que o item $(ii)$ é trivial pela contrapositiva partindo do item $(i)$.

Mostremos, por fim, o item $(iii)$. Note que:

\[
(-\lambda + \lambda)x = 0x \iff (-\lambda)x + \lambda x = 0_E \iff (-\lambda)x = -(\lambda x)
\]

A prova é análoga para $\lambda(-x) = -(\lambda x)$.

\qed

\textit{Observação: Denotaremos $(-\lambda)x$ como $-\lambda x$ e $x+(-y)$ como $x-y$.}

\section{Subespaços vetoriais}

\begin{definition}

Sejam $E$ um $\mathbb{K}$-espaço vetorial e $F\subseteq E$ um subconjunto não vazio de $E$. Dizemos que $F$ é um \textbf{subespaço vetorial de $E$} se, e somente se, $F$ é um $\mathbb{K}$-espaço vetorial com as leis interna e externa de $E$.  

\end{definition}

\hypertarget{prop221}{}
\begin{proposition}

Sejam $E$ um $\mathbb{K}$-espaço vetorial e $F\subseteq E$ um subconjunto de $E$. Logo, $F$ é um subespaço vetorial de $E$ se, e somente se:

\begin{itemize}
  \item[(i)] $F \neq \emptyset$ 
  \item[(ii)] $x,y\in F \implies x+y\in F$
  \item[(iii)] $(x,\lambda)\in F\times\mathbb{K} \implies \lambda x \in F$
\end{itemize}

\end{proposition}

\textbf{Demonstração:}

Note que, se $F$ é subespaço vetorial de $E$, então evidentemente $0_E\in F$ e logo $F\neq\emptyset$ e para quaisquer $x,y\in F$ e $\lambda\in \mathbb{K}$ temos $x+y\in F$ e $\lambda x\in \mathbb{K}$. 

Suponha, agora, que $F\neq\emptyset$ e que para quaisquer $x,y\in F$ e $\lambda\in\mathbb{K}$ temos $x+y\in F$ e $\lambda x\in F$. Note que os axiomas referentes à lei externa de $E$ são trivialmente verificados para $F$, assim como a associatividade e comutatividade para a lei interna.

Resta mostrarmos que $0_E\in F$ e que para todo $x\in F$ existe $-x\in F$ tal que $x+(-x) = x-x = 0_E$.

Como $F\neq \emptyset$, então existe pelo menos um elemento $x\in F$. Além disso, como $\lambda x\in F$ para quaisquer $\lambda\in \mathbb{K}$, temos em particular que $0x = 0_E \in F$.

Analogamente, temos diretamente que $(-1)x = -x\in F$.

Portanto, concluímos que $F$ é um subespaço vetorial de $E$.

\qed

\textit{Observação:}

\begin{proposition}

Sejam $E$ um $\mathbb{K}$-espaço vetorial e $F\subseteq E$ um subconjunto de $E$. Logo, $F$ é um subespaço vetorial de $E$ se, e somente se:

\begin{itemize}
  \item[(i)] $F \neq \emptyset$ 
  \item[(ii)] $x,y\in F, \lambda,\mu\in \mathbb{K} \implies \lambda x+\mu y\in F$
\end{itemize}

\end{proposition}

\textbf{Demonstração:}

De forma análoga à demonstração da \hyperlink{prop221}{\textbf{Proposição 2.2.1}}, se $F$ for um subespaço vetorial de $E$, então é trivial que os itens $(i)$ e $(ii)$ são satisfeitos.

Suponha, agora, que os itens $(i)$ e $(ii)$ sejam verdadeiros. Ora, nesse caso, como $F\neq\emptyset$, temos que, para qualquer $x\in F$:  

\[
0_E = 0x + 0x\in F
\]

Logo, $0_E\in F$. Ainda, note que os outros itens da \hyperlink{prop221}{\textbf{Proposição 2.2.1}} podem ser verificados facilmente. Sendo $x,y\in F$ e $\lambda\in \mathbb{K}$, temos:

\[
x + y = (1)x + (1)y \in F
\]

e, 

\[
\lambda x = \lambda x + 0_E = \lambda x + 0y\in F
\]

Logo, $F$ é subespaço vetorial de $E$.

\qed

\begin{definition}
Sejam $E$ um $\mathbb{K}$-espaço vetorial e $x_1,\ldots,x_p\in E$. Então o conjunto:

\[
\operatorname{Vet}\{x_1,\ldots,x_p\}\overset{def}{=} \{y\in E\; ;\; \exists \lambda_1,\ldots,\lambda_p \in \mathbb{K} : y = \lambda_1 x_1 + \ldots + \lambda_p x_p\}
\]

é um subespaço vetorial de $E$ denominado \textbf{subespaço gerado por} $x_1,\ldots, x_p$. Denominamos-o comumente como $\operatorname{Vet}\{x_1,\ldots,x_p\}$.

\end{definition}

Note que a construção acima é de fato um subespaço vetorial de $E$.

	Tomando $\lambda_1,\ldots,\lambda_p = 0$ temos claramente que $0_E\in \operatorname{Vet}\{x_1,\ldots,x_p\}$ e, por construção, temos que $\lambda y_1 + \mu y_2 \in \operatorname{Vet}\{x_1,\ldots,x_p\}$ para quaisquer $y_1,y_2\in \operatorname{Vet}\{x_1,\ldots,x_p\}$ e $\lambda,\mu\in \mathbb{K}$.

\begin{example}
Seja $F\subseteq \mathbb{R}^3$ munido das operações usuais de soma e multiplicação para números reais dado por:

\[
F\overset{def}{=} \{(x,y,z)\in\mathbb{R}^3 \; ; \; 2x+y+3z = 0\}
\]

Logo, $F$ é um subespaço vetorial de $\mathbb{R}^3$.
\end{example}

\textbf{Demonstração:}

Note que:

\begin{align*}
F &= \{(x,y,z) \in \mathbb{R}^3 \; ; \; 2x + y + 3z = 0\} \\
  &= \{(x,y,z) \in \mathbb{R}^3 \; ; \; y = -2x - 3z\} \\
  &= \{(x,-2x-3z,z) \in \mathbb{R}^3 \; ; \; x,z \in \mathbb{R}\} \\
  &= \{(x,-2x,0) + (0,-3z,z) \; ; \; x,z \in \mathbb{R}\} \\
  &= \{x(1, -2, 0) + z(0,-3,1) \; ; \; x,z \in \mathbb{R}\} \\
  &= \operatorname{Vet}\{(1,-2,0), (0,-3,1)\}
\end{align*}

Portanto, deduzimos que $F$ é o subespaço gerado por $(1,-2,0)$ e $(0,-3,1)$.


\begin{example}
Seja $F\subseteq \mathbb{R}^3$ munido das operações usuais de soma e multiplicação para números reais dado por:

\[
F\overset{def}{=} \{(x,y,z)\in\mathbb{R}^3 \; ; \; 2x+y+3z = 1\}
\]

Então, evidentemente $F$ não é subespaço vetorial de $\mathbb{R}^3$, pois $(0,0,0)\notin F$.
\end{example}


\begin{proposition}
Sejam $E$ um $\mathbb{K}$-espaço vetorial e $\{F_i\}_{i\in I}$ uma família de subespaços vetoriais de $E$ indexados por $I$. Então:

\begin{itemize}
  \item[(i)] $\bigcap_{i\in I} F_i$ é subespaço vetorial de $E$.  
  \item[(ii)] $\bigcup_{i\in I} F_i$ é subespaço vetorial de $E$ se $\forall i,j\in I,\; \exists k\in I$ tal que $F_i\subseteq F_k$ e $F_j\subseteq F_k$ 
  \item[(iii)] $E\setminus F_i$ não é subespaço vetorial de $E$ para qualquer $i\in I$.
\end{itemize}

\end{proposition}

\textbf{Demonstração:}

Mostremos, primeiramente, o item $(i)$. Para isso, basta mostrarmos os itens da \hyperlink{prop221}{\textbf{Proposição 2.2.1}}.

Note que $0_E\in \bigcap_{i\in I} F_i$, trivialmente, uma vez que $F_i$ é subespaço vetorial de $E$ para todo $i\in I$ e, portanto, $0_E\in F_i,\; \forall i\in I$.

Tomemos $x,y\in \bigcap_{i\in I} F_i$. Então, para todo $i\in I$, temos que $x,y\in F_i$ e, como $F_i$ é subespaço vetorial de $E$ para todo $i\in I$, deduzimos que $x+y\in F_i$, o que implica que $x+y\in \bigcap_{i\in I} F_i$.

A prova que $\lambda x \in \bigcap_{i\in I} F_i,\; \forall (\lambda, x)\in \mathbb{K}\times \bigcap_{i\in I} F_i$ é análoga.

Mostremos, agora, o item $(ii)$. Suponha inicialmente que para todo $i,j\in I,\;\exists k\in I$ tal que $F_i\subseteq F_k$ e $F_j\subseteq F_k$.

Note novamente que, trivialmente, $0_E\in \bigcup_{i\in I} F_i$.

Tomando $x,y\in\bigcup_{i\in I} F_i$, temos que $x\in F_i$ e $y\in F_j$ para algum $i,j\in I$. Como $F_i\subseteq F_k$ e $F_j\subseteq F_k$, temos portanto que $x,y\in F_k$. Ainda, como $F_k$ é subespaço vetorial de $E$, temos que $x+y\in F_k$. Portanto, concluímos que $x+y\in \bigcup_{i\in I} F_i$.

A prova que $\lambda x \in \bigcup_{i\in I} F_i, \; \forall (\lambda, x)\in \mathbb{K}\times \bigcup_{i\in I} F_i$ é análoga.

Por fim, para o item $(iii)$, note que $E\setminus F_i$ jamais será subespaço vetorial de $E$ para qualquer $i\in I$, pois $0_E \notin E\setminus F_i$. 

\qed

\section{Bases em dimensão finita}

\begin{definition}
Seja $E$ um $\mathbb{K}$-espaço vetorial. Uma família de vetores $\mathcal{F} \overset{def}{=} \{v_1,\ldots, v_n\}$ de $E$ é dita \textbf{geradora} se, e somente se, $E = \operatorname{Vet}\mathcal{F}$.
\end{definition}
